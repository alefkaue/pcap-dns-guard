cmake_minimum_required(VERSION 3.20)
project(pcap_dns_guard LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pcap_guard src/pcap_guard.cpp)
target_include_directories(pcap_guard PUBLIC include)
add_executable(pcap-dns-guard src/main.cpp)
target_link_libraries(pcap-dns-guard PRIVATE pcap_guard)
add_executable(guard-tests tests/tests.cpp)
target_link_libraries(guard-tests PRIVATE pcap_guard)
if(MSVC)
  target_compile_options(pcap_guard PRIVATE /W4 /WX)
else()
  target_compile_options(pcap_guard PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(ENABLE_SANITIZERS AND NOT MSVC)
  foreach(t pcap_guard pcap-dns-guard guard-tests)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
add_test(NAME parser-regressions COMMAND guard-tests)
